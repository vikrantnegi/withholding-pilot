SELECT name, price FROM products WHERE price > 5000 ORDER BY price DESC;
-- Q3
SELECT city, COUNT(*) FROM customers GROUP BY city;
-- Q4
SELECT category, AVG(price) FROM products GROUP BY category;
-- Q5
SELECT o.id, c.name, o.order_date FROM orders o, customers c WHERE c.id=o.customer_id AND o.status='completed';
-- Q6
SELECT p.name, oi.quantity FROM order_items oi JOIN products p ON p.id=oi.product_id WHERE oi.order_id=3;
-- Q7
SELECT c.name, SUM(oi.quantity*oi.unit_price) FROM customers c JOIN orders o ON o.customer_id=c.id JOIN order_items oi ON oi.order_id=o.id GROUP BY c.name;
-- Q8
SELECT name FROM customers WHERE id NOT IN (SELECT customer_id FROM orders);
